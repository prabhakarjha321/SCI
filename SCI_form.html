    <script>

        document.getElementById('personalInfoForm').addEventListener('submit', function(event) {

            let hasErrors = false;


            // Reset all error messages

            document.querySelectorAll('.error-message').forEach(node => node.textContent = '');


            // Name validation

            const nameInput = document.getElementById('name');

            if (!nameInput.value.trim()) {

                document.getElementById('name-error').textContent = 'Name is required';

                hasErrors = true;

            }


            // Email validation

            const emailInput = document.getElementById('email');

            const emailRegex = /^[\w-\.]+@([\w-]+\.)+[\w-]{2,4}$/;

            if (!emailInput.value.trim()) {

                document.getElementById('email-error').textContent = 'Email is required';

                hasErrors = true;

            } else if (!emailRegex.test(emailInput.value)) {

                document.getElementById('email-error').textContent = 'Invalid email format';

                hasErrors = true;

            }


            // Age validation

            const ageInput = document.getElementById('age');

            const ageValue = parseInt(ageInput.value);

            if (!ageInput.value.trim()) {

                document.getElementById('age-error').textContent = 'Age is required';

                hasErrors = true;

            } else if (isNaN(ageValue) || ageValue <= 0) {

                document.getElementById('age-error').textContent = 'Invalid age';

                hasErrors = true;

            }


            // Phone validation

            const phoneInput = document.getElementById('phone');

            const phoneRegex = /^\d{10}$/; // Basic 10-digit phone number validation

            if (!phoneInput.value.trim()) {

                document.getElementById('phone-error').textContent = 'Phone number is required';

                hasErrors = true;

            } else if (!phoneRegex.test(phoneInput.value)) {

                document.getElementById('phone-error').textContent = 'Invalid phone number format (10 digits required)';

                hasErrors = true;

            }


            // Address validation

            const addressInput = document.getElementById('address');

            if (!addressInput.value.trim()) {

                document.getElementById('address-error').textContent = 'Address is required';

                hasErrors = true;

            }


            // SSN validation

            const ssnInput = document.getElementById('ssn');

            const ssnRegex = /^\d{3}-\d{2}-\d{4}$/;

            if (!ssnInput.value.trim()) {

                document.getElementById('ssn-error').textContent = 'Social Security Number is required';

                hasErrors = true;

            } else if (!ssnRegex.test(ssnInput.value)) {

                document.getElementById('ssn-error').textContent = 'Invalid SSN format (XXX-XX-XXXX)';

                hasErrors = true;

            }



            if (hasErrors) {

                event.preventDefault(); // Prevent form submission if there are errors

            } else {

                // If all fields are valid, you can proceed with form submission

                // In a real application, you would send this data to a server

                console.log('Form submitted successfully!');

                alert('Form submitted successfully! (Data not actually sent)');

            }

        });
